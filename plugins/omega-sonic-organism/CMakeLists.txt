cmake_minimum_required(VERSION 3.25)
project(OmegaSonicOrganismCore VERSION 1.0.0 LANGUAGES C CXX)
# Ship a self-contained Windows DLL rather than relying on the host's CRT DLL
# versions. The same policy must reach SDK libraries and the portable core.
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(SMTG_USE_STATIC_CRT ON CACHE BOOL "Use static Microsoft runtime" FORCE)
option(OSO_BUILD_PLUGIN "Build Steinberg Windows VST3 instrument" OFF)
set(OSO_SDK_ROOT "" CACHE PATH "Pinned official Steinberg VST3 SDK")
include(CTest)
add_library(omega_organism_core STATIC core/src/InstrumentEngine.cpp core/src/Preset.cpp)
target_compile_features(omega_organism_core PUBLIC cxx_std_17)
target_include_directories(omega_organism_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/core/include)
if(MSVC)
  target_compile_options(omega_organism_core PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(omega_organism_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(BUILD_TESTING)
  add_executable(omega_core_tests tests/core_tests.cpp)
  target_link_libraries(omega_core_tests PRIVATE omega_organism_core)
  add_test(NAME OmegaCoreTests COMMAND omega_core_tests)
  add_executable(omega_state_tests tests/state_tests.cpp)
  target_link_libraries(omega_state_tests PRIVATE omega_organism_core)
  add_test(NAME OmegaStateTests COMMAND omega_state_tests)
  add_executable(omega_engine_tests tests/engine_tests.cpp)
  target_link_libraries(omega_engine_tests PRIVATE omega_organism_core)
  add_test(NAME OmegaEngineTests COMMAND omega_engine_tests)
  add_executable(omega_engine_metrics tools/engine_metrics.cpp)
  target_link_libraries(omega_engine_metrics PRIVATE omega_organism_core)
  add_executable(omega_cpu_profile tools/cpu_profile.cpp)
  target_link_libraries(omega_cpu_profile PRIVATE omega_organism_core)
  add_executable(omega_preset_tests tests/preset_tests.cpp)
  target_link_libraries(omega_preset_tests PRIVATE omega_organism_core)
  add_test(NAME OmegaPresetTests COMMAND omega_preset_tests)
  add_executable(omega_calibrate_bank tools/calibrate_bank.cpp)
  target_link_libraries(omega_calibrate_bank PRIVATE omega_organism_core)
endif()

if(OSO_BUILD_PLUGIN)
  if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The first target is Windows x64")
  endif()
  if(NOT EXISTS "${OSO_SDK_ROOT}/public.sdk/source/vst/vstaudioeffect.h")
    message(FATAL_ERROR "Set OSO_SDK_ROOT to the official SDK")
  endif()
  set(SMTG_ENABLE_VSTGUI_SUPPORT OFF CACHE BOOL "" FORCE)
  set(SMTG_ENABLE_VST3_PLUGIN_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(SMTG_ENABLE_VST3_HOSTING_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(SMTG_CREATE_PLUGIN_LINK OFF CACHE BOOL "" FORCE)
  set(SMTG_RUN_VST_VALIDATOR OFF CACHE BOOL "" FORCE)
  add_subdirectory("${OSO_SDK_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/sdk")
  # SDK is a subdirectory: initialize its bundle configurations and entry-point
  # source path in this directory too. Keep its /fp:fast defaults out of DSP.
  smtg_configure_cmake_generator()
  set(public_sdk_SOURCE_DIR "${OSO_SDK_ROOT}/public.sdk")
  # Official VSTGUI Windows amalgamation: avoids building unused XML editor,
  # live-editing tools and hundreds of separate translation units. No SDK edits.
  add_library(oso_vstgui STATIC "${OSO_SDK_ROOT}/vstgui4/vstgui/vstgui_win32.cpp")
  target_include_directories(oso_vstgui PUBLIC "${OSO_SDK_ROOT}/vstgui4")
  target_compile_features(oso_vstgui PUBLIC cxx_std_17)
  target_compile_definitions(oso_vstgui PUBLIC UNICODE _UNICODE NOMINMAX VSTGUI_ENABLE_DEPRECATED_METHODS=1)
  target_compile_options(oso_vstgui PRIVATE /EHa /bigobj /utf-8)
  target_link_libraries(oso_vstgui PUBLIC d2d1 dwrite windowscodecs shlwapi comctl32 ole32 uuid dwmapi d3d11 dcomp dxgi opengl32)
  smtg_add_vst3plugin(OmegaSonicOrganism plugin/Plugin.cpp plugin/Plugin.h plugin/Editor.cpp plugin/Editor.h
    "${OSO_SDK_ROOT}/public.sdk/source/vst/vstguieditor.cpp"
    "${OSO_SDK_ROOT}/public.sdk/source/vst/vstgui_win32_bundle_support.cpp")
  target_include_directories(OmegaSonicOrganism PRIVATE "${OSO_SDK_ROOT}")
  target_link_libraries(OmegaSonicOrganism PRIVATE sdk oso_vstgui omega_organism_core)
  target_compile_definitions(OmegaSonicOrganism PRIVATE OSO_WITH_EDITOR=1 SMTG_MODULE_IS_BUNDLE=1)
  target_compile_features(OmegaSonicOrganism PRIVATE cxx_std_17)
  target_compile_options(OmegaSonicOrganism PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/utf-8>)
  if(BUILD_TESTING)
    add_executable(omega_adapter_tests tests/adapter_tests.cpp plugin/Plugin.cpp "${OSO_SDK_ROOT}/public.sdk/source/common/memorystream.cpp")
    target_include_directories(omega_adapter_tests PRIVATE "${OSO_SDK_ROOT}" plugin)
    target_link_libraries(omega_adapter_tests PRIVATE sdk sdk_hosting omega_organism_core)
    target_compile_options(omega_adapter_tests PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/utf-8>)
    add_test(NAME OmegaAdapterTests COMMAND omega_adapter_tests)
    add_executable(omega_binary_probe tools/binary_probe.cpp "${OSO_SDK_ROOT}/public.sdk/source/common/memorystream.cpp")
    target_include_directories(omega_binary_probe PRIVATE "${OSO_SDK_ROOT}" plugin)
    target_link_libraries(omega_binary_probe PRIVATE sdk sdk_hosting omega_organism_core)
    target_compile_options(omega_binary_probe PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/utf-8>)
    add_executable(omega_audition_probe tools/audition_probe.cpp "${OSO_SDK_ROOT}/public.sdk/source/common/memorystream.cpp")
    target_include_directories(omega_audition_probe PRIVATE "${OSO_SDK_ROOT}" plugin)
    target_link_libraries(omega_audition_probe PRIVATE sdk sdk_hosting omega_organism_core)
    target_compile_options(omega_audition_probe PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/utf-8>)
    add_executable(omega_editor_probe tools/editor_probe.cpp)
    target_include_directories(omega_editor_probe PRIVATE "${OSO_SDK_ROOT}" plugin)
    target_link_libraries(omega_editor_probe PRIVATE sdk sdk_hosting omega_organism_core ole32 user32 gdi32)
    target_compile_options(omega_editor_probe PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/utf-8>)
  endif()
endif()
