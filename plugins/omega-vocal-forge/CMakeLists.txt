cmake_minimum_required(VERSION 3.25)
project(OmegaVocalForge VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(SMTG_USE_STATIC_CRT ON CACHE BOOL "" FORCE)
include(CTest)
option(OVF_BUILD_PLUGIN "Build Windows x64 VST3" ON)
set(OVF_SDK_ROOT "" CACHE PATH "Pinned Steinberg SDK 3.8.1")
add_executable(ovf_engine_tests tests/EngineTests.cpp)
target_include_directories(ovf_engine_tests PRIVATE src)
add_test(NAME VocalEngine COMMAND ovf_engine_tests)
if(OVF_BUILD_PLUGIN)
 if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 required")
 endif()
 if(NOT EXISTS "${OVF_SDK_ROOT}/LICENSE.txt")
  message(FATAL_ERROR "Supply -DOVF_SDK_ROOT=<Steinberg SDK 3.8.1>; see dependencies.lock.json")
 endif()
 set(SMTG_ENABLE_VSTGUI_SUPPORT OFF CACHE BOOL "" FORCE)
 set(SMTG_ENABLE_VST3_PLUGIN_EXAMPLES OFF CACHE BOOL "" FORCE)
 set(SMTG_ENABLE_VST3_HOSTING_EXAMPLES OFF CACHE BOOL "" FORCE)
 set(SMTG_CREATE_PLUGIN_LINK OFF CACHE BOOL "" FORCE)
 set(SMTG_RUN_VST_VALIDATOR OFF CACHE BOOL "" FORCE)
 add_subdirectory("${OVF_SDK_ROOT}" sdk)
 smtg_configure_cmake_generator()
 set(public_sdk_SOURCE_DIR "${OVF_SDK_ROOT}/public.sdk")
 smtg_add_vst3plugin(OmegaVocalForge src/Plugin.cpp src/Editor.cpp)
 target_include_directories(OmegaVocalForge PRIVATE "${OVF_SDK_ROOT}" src)
 target_compile_definitions(OmegaVocalForge PRIVATE UNICODE _UNICODE NOMINMAX SMTG_MODULE_IS_BUNDLE=1)
 target_link_libraries(OmegaVocalForge PRIVATE sdk comctl32 user32 gdi32)
 add_executable(ovf_adapter_tests tests/AdapterTests.cpp src/Plugin.cpp src/Editor.cpp "${OVF_SDK_ROOT}/public.sdk/source/common/memorystream.cpp")
 target_include_directories(ovf_adapter_tests PRIVATE "${OVF_SDK_ROOT}" src)
 target_compile_definitions(ovf_adapter_tests PRIVATE OVF_NO_FACTORY=1 UNICODE _UNICODE NOMINMAX)
 target_link_libraries(ovf_adapter_tests PRIVATE sdk sdk_hosting comctl32 user32 gdi32)
 add_test(NAME VocalAdapter COMMAND ovf_adapter_tests)
 add_executable(ovf_artifact_probe tools/ArtifactProbe.cpp "${OVF_SDK_ROOT}/public.sdk/source/common/memorystream.cpp")
 target_include_directories(ovf_artifact_probe PRIVATE "${OVF_SDK_ROOT}" src)
 target_compile_definitions(ovf_artifact_probe PRIVATE UNICODE _UNICODE NOMINMAX)
 target_link_libraries(ovf_artifact_probe PRIVATE sdk sdk_hosting ole32)
endif()
