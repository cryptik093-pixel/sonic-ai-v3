cmake_minimum_required(VERSION 3.24)
project(ADLIB93 VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(MSVC)
    add_compile_options(/Zc:__cplusplus /W4)
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "ADLIB-93 requires a 64-bit build")
endif()
option(ADLIB93_BUILD_PLUGIN "Build Windows VST3 and native editor" OFF)
include(CTest)
file(GLOB_RECURSE ADLIB_DSP CONFIGURE_DEPENDS src/core/*.cpp src/dsp/*.cpp src/presets/*.cpp)
add_library(adlib93_dsp STATIC ${ADLIB_DSP})
target_include_directories(adlib93_dsp PUBLIC src)
file(GLOB ADLIB_TESTS CONFIGURE_DEPENDS tests/*Tests.cpp)
add_executable(adlib93_tests tests/TestMain.cpp tests/TestSupport.cpp ${ADLIB_TESTS})
target_link_libraries(adlib93_tests PRIVATE adlib93_dsp)
add_test(NAME Build.Configuration COMMAND adlib93_tests --case Build.Configuration)
add_test(NAME Build.UnknownCase COMMAND adlib93_tests --case nonexistent)
set_tests_properties(Build.UnknownCase PROPERTIES WILL_FAIL TRUE)
