function(adlib_register_cases target)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${ARGN})
    foreach(source IN LISTS ARGN)
        file(READ "${source}" content)
        string(REGEX MATCHALL "TEST_CASE\\(\"[^\"]+\"\\)" matches "${content}")
        foreach(match IN LISTS matches)
            string(REGEX REPLACE "TEST_CASE\\(\"([^\"]+)\"\\)" "\\1" name "${match}")
            add_test(NAME "${name}" COMMAND ${target} --case "${name}")
        endforeach()
    endforeach()
endfunction()
adlib_register_cases(adlib93_tests ${ADLIB_TESTS})
add_test(NAME Build.UnknownCase COMMAND adlib93_tests --case nonexistent)
set_tests_properties(Build.UnknownCase PROPERTIES WILL_FAIL TRUE)

foreach(tool IN ITEMS RenderFixtures Benchmark)
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tools/${tool}.cpp")

            set(tool_target adlib93_render)
        else()
            set(tool_target adlib93_benchmark)
        endif()
        add_executable(${tool_target} tools/${tool}.cpp)
        target_link_libraries(${tool_target} PRIVATE adlib93_dsp)
    endif()
endforeach()
